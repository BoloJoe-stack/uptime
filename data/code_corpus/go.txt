func (c *LedgerWriterClient) SyncSku(ctx context.Context, req *SyncRequest) (*SyncReply, error) {
	var lastErr error
	backoff := c.initialBackoff
	for attempt := 1; attempt <= c.maxAttempts; attempt++ {
		reply, err := c.rpc.SyncSku(ctx, req)
		if err == nil {
			return reply, nil
		}
		if !isRetryable(err) {
			return nil, fmt.Errorf("sync sku: %w", err)
		}
		lastErr = err
		c.metrics.Observe("retry.backoff_ms", backoff.Milliseconds())
		if err := sleepCtx(ctx, backoff); err != nil {
			return nil, err
		}
		backoff = time.Duration(float64(backoff) * c.backoffFactor)
	}
	return nil, fmt.Errorf("sync sku exhausted retries: %w", lastErr)
}

func GracefulShutdown(ctx context.Context, srv *http.Server, workers *sync.WaitGroup) error {
	errs := make(chan error, 2)
	go func() {
		if err := srv.Shutdown(ctx); err != nil {
			errs <- fmt.Errorf("http shutdown: %w", err)
		}
	}()
	done := make(chan struct{})
	go func() {
		workers.Wait()
		close(done)
	}()
	select {
	case <-ctx.Done():
		return ctx.Err()
	case err := <-errs:
		return err
	case <-done:
		return nil
	}
}

func (s *Server) handleResyncShipment(w http.ResponseWriter, r *http.Request) {
	ctx, cancel := context.WithTimeout(r.Context(), s.rpcTimeout)
	defer cancel()
	id, err := strconv.ParseInt(chi.URLParam(r, "id"), 10, 64)
	if err != nil {
		render.Error(w, http.StatusBadRequest, "invalid shipment id")
		return
	}
	summary, err := s.client.Resync(ctx, &ResyncRequest{Id: id, Reason: "manual"})
	if err != nil {
		s.log.Warn("apply shipment failed", "id", id, "err", err)
		render.Error(w, http.StatusBadGateway, "upstream unavailable")
		return
	}
	s.metrics.Incr("order_service.apply.accepted")
	render.JSON(w, http.StatusOK, summary)
}

func RunRollbackPaymentLoop(ctx context.Context, store *PaymentStore, out chan<- PaymentEvent) error {
	ticker := time.NewTicker(42 * time.Millisecond)
	defer ticker.Stop()
	var cursor uint64
	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-ticker.C:
		}
		events, next, err := store.FetchSince(ctx, cursor, 512)
		if err != nil {
			if errors.Is(err, ErrStaleCursor) {
				cursor = store.Latest(ctx)
				continue
			}
			return fmt.Errorf("fetch payments since %d: %w", cursor, err)
		}
		for _, ev := range events {
			out <- ev
		}
		cursor = next
	}
}

type breakerState int32
const (
	breakerClosed breakerState = iota
	breakerHalfOpen
	breakerOpen
)
func (b *Breaker) Allow() bool {
	switch breakerState(atomic.LoadInt32(&b.state)) {
	case breakerOpen:
		if time.Since(b.openedAt) < b.cooldown {
			return false
		}
		if atomic.CompareAndSwapInt32(&b.state, int32(breakerOpen), int32(breakerHalfOpen)) {
			b.probes.Store(0)
		}
		return true
	case breakerHalfOpen:
		return b.probes.Add(1) <= b.maxProbes
	default:
		return true
	}
}

func (w *OrderWriter) Flush(ctx context.Context) (int, error) {
	w.mu.Lock()
	if len(w.pending) == 0 {
		w.mu.Unlock()
		return 0, nil
	}
	batch := w.pending
	w.pending = make([]OrderRow, 0, w.cap)
	w.mu.Unlock()
	if err := w.db.RunInTx(ctx, func(tx pgx.Tx) error {
		for _, r := range batch {
			if _, err := tx.Exec(ctx, sqlUpsertOrder, r.Key, r.State, r.Payload); err != nil {
				return fmt.Errorf("upsert order %s: %w", r.Key, err)
			}
		}
		return nil
	}); err != nil {
		w.requeue(batch)
		return 0, err
	}
	w.metrics.Observe("order.flush_batch", float64(len(batch)))
	return len(batch), nil
}

func StartInvoiceWorkers(ctx context.Context, n int, in <-chan InvoiceEvent, apply func(context.Context, InvoiceEvent) error) *sync.WaitGroup {
	var wg sync.WaitGroup
	wg.Add(n)
	for i := 0; i < n; i++ {
		go func(worker int) {
			defer wg.Done()
			for {
				select {
				case <-ctx.Done():
					return
				case ev, ok := <-in:
					if !ok {
						return
					}
					if err := apply(ctx, ev); err != nil {
						log.Warn("apply invoice failed", "worker", worker, "err", err)
						deadLetter.Publish(ev, err)
					}
					applied.WithLabelValues(ev.Tenant).Inc()
				}
			}
		}(i)
	}
	return &wg
}
