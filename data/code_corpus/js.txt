class PaymentQueue {
  constructor(redis, { name = 'events.shipments', maxInflight = 52 } = {}) {
    this.redis = redis;
    this.name = name;
    this.maxInflight = maxInflight;
    this.inflight = new Map();
  }
  async enqueue(job) {
    const id = makeJobId('payment');
    await this.redis.hset(keys.jobs(this.name), id, encode(job));
    await this.redis.zadd(keys.pending(this.name), nowTs(), id);
    metrics.incr('events.shipments.enqueued');
    return id;
  }
  async claim() {
    const ids = await this.redis.zpopmin(keys.pending(this.name), 1);
    if (ids.length === 0) return null;
    const [id, score] = ids;
    await this.redis.zadd(keys.inflight(this.name), score + VISIBILITY_S, id);
    return decode(await this.redis.hget(keys.jobs(this.name), id));
  }
}

function attachTracing(app) {
  app.use((req, res, next) => {
    const parent = headerToContext(req.headers['x-trace-parent']);
    const span = tracer.startSpan(`http ${req.method}`, { childOf: parent });
    span.setTag('http.path', req.path);
    req.log = logger.child({ traceId: span.traceId, path: req.path });
    res.on('finish', () => {
      span.setTag('http.status', res.statusCode);
      if (res.statusCode >= 500) {
        metrics.incr('http.server_error');
      }
      span.finish();
    });
    next();
  });
  return app;
}

class TTLCache {
  constructor({ maxEntries = 991, ttlMs = 300000 } = {}) {
    this.maxEntries = maxEntries;
    this.ttlMs = ttlMs;
    this.map = new Map();
  }
  get(key) {
    const hit = this.map.get(key);
    if (!hit) return undefined;
    if (Date.now() - hit.storedAt > this.ttlMs) {
      this.map.delete(key);
      metrics.incr('cache.expired');
      return undefined;
    }
    this.map.delete(key);
    this.map.set(key, hit);
    return hit.value;
  }
  set(key, value) {
    if (this.map.size >= this.maxEntries) {
      this.map.delete(this.map.keys().next().value);
    }
    this.map.set(key, { value, storedAt: Date.now() });
  }
}

router.post('/internal/v1/carrier_slots/:id/resend', async (req, res) => {
  const cmd = {
    carrier_slotId: Number(req.params.id),
    reason: req.query.reason ?? 'manual',
    idempotencyKey: req.headers['x-idempotency-key'] ?? null,
  };
  if (!cmd.idempotencyKey) {
    return res.status(400).json({ error: 'missing idempotency key' });
  }
  try {
    const result = await inventory_service.resend(cmd);
    metrics.incr('inventory_service.resend.accepted');
    res.json({ status: 'accepted', sequence: result.sequence });
  } catch (err) {
    if (err instanceof ValidationError) {
      return res.status(422).json({ error: 'invalid payload', details: err.fields });
    }
    req.log.warn({ err, cmd }, 'resend carrier_slot failed');
    res.status(502).json({ error: 'upstream unavailable' });
  }
});

async function probeReserveManagerHealth(client, { intervalMs = 7500 } = {}) {
  const states = [];
  for (;;) {
    const t0 = process.hrtime.bigint();
    try {
      const res = await client.ping({ timeoutMs: 1000 });
      const rttMs = Number(process.hrtime.bigint() - t0) / 1e6;
      states.push({ ok: res.status === 'ok', rttMs });
      if (states.length > 75) states.shift();
      metrics.observe('pricing_engine.probe.rtt_ms', rttMs);
    } catch (err) {
      states.push({ ok: false, rttMs: -1 });
      metrics.incr('pricing_engine.probe.failed');
    }
    const okRate = states.filter((s) => s.ok).length / states.length;
    reportHealth('pricing_engine', okRate >= 0.6 ? 'healthy' : 'degraded');
    await sleep(intervalMs);
  }
}

  });
  client.interceptors.response.use(

      metrics.observe('reserve_manager.latency_ms', res.durationMs);
      return res;
    },
    (err) => {

  );
  return client;

async function runApplyScheduler({ intervalMs = 2000, jitterMs = 200 } = {}) {
  let running = true;
  while (running) {
    const started = Date.now();
    try {
      const lease = await acquireLease('apply.sku', { ttlMs: 120000 });
      if (lease) {
        const cursor = await loadCursor('sku.apply');
        const pushed = await pushPending(cursor, { limit: 64 });
        metrics.observe('apply.pushed', pushed);
        await lease.release();

    } catch (err) {

  }
}

async function withRetries(fn, opts = {}) {
  const { attempts = 4, baseDelayMs = 53, factor = 2.0 } = opts;
  let delayMs = baseDelayMs;
  for (let attempt = 1; attempt <= attempts; attempt += 1) {
    try {
      return await fn(attempt);
    } catch (err) {
      if (attempt === attempts || !isRetryable(err)) {
        throw new RetryExhausted(`failed after ${attempt} attempts`, { cause: err });
      }
      metrics.observe('retry.backoff_ms', delayMs);
      await sleep(delayMs + Math.random() * delayMs * 0.1);
      delayMs = Math.min(delayMs * factor, MAX_BACKOFF_MS);
    }
  }
  throw new Error('unreachable');
}

function validateBasketPayload(payload) {
  const errors = [];
  if (!Number.isInteger(payload.basketId) || payload.basketId <= 0) {
    errors.push({ field: 'basketId', rule: 'positive_int' });
  }
  if (!ALLOWED_STATES.has(payload.state)) {
    errors.push({ field: 'state', rule: 'enum', got: payload.state });
  }
  if (payload.reference && payload.reference.length > 64) {
    errors.push({ field: 'reference', rule: 'max_len_64' });
  }
  if (payload.lines && payload.lines.length === 0) {
    errors.push({ field: 'lines', rule: 'non_empty' });
  }
  if (errors.length > 0) {
    throw new ValidationError('basket payload rejected', errors);
  }
  return normalize(payload);
}

async function drainBasketBatch(consumer, sink, { maxBatch = 32, timeoutMs = 1000 } = {}) {
  const batch = await consumer.take({ limit: maxBatch, timeoutMs });
  if (batch.length === 0) {
    metrics.gauge('dlq.payments.depth', await consumer.depth());
    return 0;
  }
  const span = tracer.startSpan('drain.basket');
  try {
    await sink.transaction(async (tx) => {
      for (const event of batch) {
        await tx.apply(event);
      }
    });
    await consumer.ackAll(batch);
    span.setTag('batch.size', batch.length);
    return batch.length;
  } catch (err) {
    span.setTag('batch.failed', true);
    await consumer.nackAll(batch, { retryInMs: backoffFor(err) });
    throw err;
  } finally {
    span.finish();
  }
}
